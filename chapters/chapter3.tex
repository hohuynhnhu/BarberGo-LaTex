\subsubsection{Xác thực}
\begin{itemize}
    \item \textbf{Đăng ký tài khoản}
    \begin{itemize}
        \item Người dùng có thể tạo tài khoản mới bằng email hoặc tài khoản Google
        \item Hệ thống yêu cầu xác thực email trước khi kích hoạt tài khoản
    \end{itemize}
    
    \item \textbf{Đăng nhập}
    \begin{itemize}
        \item Hỗ trợ đăng nhập bằng email và mật khẩu
        \item Tích hợp đăng nhập qua Google.
    \end{itemize}
    
    \item \textbf{Quên mật khẩu}
    \begin{itemize}
        \item Cho phép người dùng đặt lại mật khẩu qua email.
        \item Hệ thống gửi email để xác thực danh tính
    \end{itemize}
    
    \item \textbf{Đổi mật khẩu}
    \begin{itemize}
        \item Người dùng có thể thay đổi mật khẩu trong phần cài đặt tài khoản
        \item Yêu cầu xác thực mật khẩu cũ trước khi cho phép thay đổi
    \end{itemize}
    
    \item \textbf{Đăng xuất}
    \begin{itemize}
        \item Cho phép người dùng đăng xuất khỏi ứng dụng
        \item Xóa token xác thực và dữ liệu phiên làm việc
    \end{itemize}
    
\end{itemize}

\subsubsection{Quản lý đặt lịch}
\begin{itemize}
    \item \textbf{Tạo đặt lịch}
    \begin{itemize}
        \item Người dùng chọn của tiệm, dịch vụ và thời gian mong muốn
        \item Hiển thị các khung giờ còn trống theo thời gian 9thực
    \end{itemize}
    
    \item \textbf{Xem lịch đặt}
    \begin{itemize}
        \item Hiển thị danh sách các lịch hẹn đã đặt (quá khứ, hiện tại, tương lai)
        \item Phân loại theo trạng thái: đã xác nhận, sắp đến, đã hoàn thành, đã hủy
        \item Cung cấp bộ lọc và tìm kiếm lịch hẹn
    \end{itemize}
    
    \item \textbf{Sửa lịch đặt}
    \begin{itemize}
        \item Cho phép thay đổi thời gian, dịch vụ và cửa tiệm
    \end{itemize}
    
    \item \textbf{Hủy lịch đặt}
    \begin{itemize}
        \item Cho phép hủy lịch hẹn 
        \item Thông báo cho cửa tiệm khi lịch hẹn bị hủy
    \end{itemize}
\end{itemize}

\subsubsection{Hồ sơ cá nhân}
\begin{itemize}
    \item \textbf{Cập nhật thông tin}
    \begin{itemize}
        \item Chỉnh sửa thông tin cá nhân: tên, số điện thoại, email
        \item Upload và thay đổi ảnh đại diện
    \end{itemize}
    
    \item \textbf{Xem hồ sơ}
    \begin{itemize}
        \item Hiển thị đầy đủ thông tin tài khoản
        \item Thống kê số lần sử dụng dịch vụ, điểm thưởng
        \item Xem lịch sử giao dịch và lịch hẹn
    \end{itemize}
\end{itemize}

\subsubsection{Đánh giá}
\begin{itemize}
    \item \textbf{Đánh giá sao}
    \begin{itemize}
        \item Đánh giá dịch vụ bằng thang điểm 1-5 sao
        \item Đánh giá cửa cửa hàng.
    \end{itemize}
    
    \item \textbf{Xem lịch sử đánh giá}
    \begin{itemize}
        \item Hiển thị các đánh giá đã gửi trước đó
        \item Cho phép chỉnh sửa hoặc xóa đánh giá trong thời gian cho phép
    \end{itemize}
\end{itemize}

\subsubsection{Điều hướng}
\begin{itemize}
    \item \textbf{Xem bản đồ}
    \begin{itemize}
        \item Hiển thị vị trí cửa tiệm trên bản đồ
        \item Tích hợp OpenStreetMap
    \end{itemize}
    
    \item \textbf{Chỉ đường đến tiệm}
    \begin{itemize}
        \item Cung cấp hướng dẫn đường đi từ vị trí hiện tại đến cửa tiệm
        \item Hiển thị thời gian di chuyển ước tính
    \end{itemize}
\end{itemize}

\subsubsection{ Thông báo}
\begin{itemize}
    \item \textbf{Nhắc hẹn sắp tới}
    \begin{itemize}
        \item Gửi thông báo sắp tới lịch hẹn
        \item Thông báo khi cửa tiệm từ chối hoặc hoàn thành lịch hẹn
    \end{itemize}
\end{itemize}

\section{Yêu cầu Phi chức năng}
\subsection{Hiệu năng (Performance)}
\begin{itemize}
    \item Thời gian phản hồi thao tác thông thường: 2-3 giây
    \item Thời gian xử lý AI tạo kiểu tóc: 10-15 giây
    \item Thời gian xử lý AI phân tích mụn: 3-5 giây
    \item Thời gian phản hồi chatbot RAG: 2-4 giây
    \item Thời gian load danh sách và báo cáo: 2-5 giây
\end{itemize}

\subsection{Khả năng mở rộng (Scalability)}
\begin{itemize}
    \item Kiến trúc phân lớp Three-Tier cho phép mở rộng độc lập từng tầng
    \item Backend services tách biệt theo chức năng
    \item Database schema thiết kế hỗ trợ mở rộng features
    \item Component-based architecture cho frontend
    \item API RESTful chuẩn hóa, dễ tích hợp
\end{itemize}

\subsection{Bảo mật (Security)}
\begin{itemize}
    \item Xác thực qua Supabase Authentication (Email/Password, Google OAuth)
    \item JWT tokens tự động quản lý bởi Supabase
    \item Row Level Security (RLS) kiểm soát quyền truy cập dữ liệu
    \item Pydantic schemas validation cho input
    \item Environment variables cho API keys
\end{itemize}

\textbf{Lưu ý:} Sản phẩm demo chạy môi trường local, chưa triển khai các biện pháp bảo mật nâng cao cho production.

\subsection{Tính khả dụng (Availability)}
\begin{itemize}
    \item Xử lý lỗi với try-catch và error logging
    \item Thông báo lỗi user-friendly
    \item Cache dữ liệu cơ bản trên mobile app
    \item Retry mechanism cho API calls thất bại
\end{itemize}

\subsection{Khả năng bảo trì (Maintainability)}
\begin{itemize}
    \item Code tuân thủ coding standards (PEP 8, Dart style guide, Vue.js style guide)
    \item Kiến trúc phân lớp rõ ràng
    \item Comments và docstrings cho code quan trọng
    \item Version control với Git
    \item Separation of concerns, reusable components
\end{itemize}

\subsection{Khả năng kiểm thử (Testability)}
\begin{itemize}
    \item Manual testing cho chức năng chính
    \item Postman collection để test APIs
    \item Test cases cho user flows quan trọng
    \item Mock data cho UI/UX testing
\end{itemize}
